{
  const width = 900;
  const height = 700;

  const div = d3.create("div")
      .style("width", "100%")
      .style("height", height + "px")
      .style("position", "relative");

  const svg = div.append("svg")
      .attr("width", "100%")
      .attr("height", height)
      .style("cursor", "grab");

  // Tooltip
  const tooltip = div.append("div")
      .style("position", "absolute")
      .style("pointer-events", "none")
      .style("background", "white")
      .style("padding", "6px 10px")
      .style("border", "1px solid #ccc")
      .style("border-radius", "4px")
      .style("font-size", "12px")
      .style("opacity", 0);

  // Projection centered on your Leaflet map center
  const projection = d3.geoMercator()
      .center([-112, 36.5])
      .scale(2000)
      .translate([width / 2, height / 2]);

  const path = d3.geoPath().projection(projection);

  // Color scale
  const color_scale = d3.scaleSequential()
      .domain([0, 30])
      .interpolator(d3.interpolateRdYlGn);

  // Draw polygons
  const g = svg.append("g");
  
  // Watershed basins
  g.selectAll("path")
    .data(HUC6_geo.features)
    .join("path")
      .attr("d", path)
      .attr("stroke", "black")
      .attr("stroke-width", 1)
      .attr("fill", d => color_scale(d.properties.Average_SWE))
      .attr("fill-opacity", 0.4)
      .on("mousemove", (event, d) => {
        tooltip.style("opacity", 1)
          .style("left", event.offsetX + 15 + "px")
          .style("top", event.offsetY + "px")
          .html(`<strong>${d.properties.name}</strong><br>${d.properties.Average_SWE} in`);
      })
      .on("mouseleave", () => tooltip.style("opacity", 0));

  // Zoom behavior
  svg.call(
    d3.zoom()
      .scaleExtent([1, 15])
      .on("zoom", (event) => {
        g.attr("transform", event.transform);
      })
  );

  // Legend
  const legend = div.append("div")
      .style("position", "absolute")
      .style("bottom", "20px")
      .style("left", "20px")
      .style("background", "white")
      .style("padding", "10px 14px")
      .style("border-radius", "6px")
      .style("border", "1px solid #dee2e6")
      .style("font-size", "12px")
      .style("font-family", "sans-serif");

  legend.html(`
    <div style="
      background: linear-gradient(to left, green, yellow, red);
      height: 20px; width: 200px; margin-bottom: 4px;">
    </div>
    <div style="display: flex; justify-content: space-between;">
      <span>0 in</span><span>30 in</span>
    </div>
  `);

  return div.node();
}
