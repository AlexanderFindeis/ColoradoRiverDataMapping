Surface water extent data is from the JRC Global Surface Water Dataset, using satellite data to map surface water from 1984 to 2020. Years after 2020 are created using data from the Sentinel-2 satellite through the OpenEO API. Normalized Difference Water Index was calculated and vectorized to create polygons of surface water extent for included reservoirs, using the median value for each pixel across the entire year. The current year is the median of the past two months to approximate current levels and account for discrepencies and cloud cover in individual collections.

// Date slider

viewof DateSlider = {
  // Initial bounds come from the full, unfiltered data (static)
  let min_ts = d3.min(Res_Elv, d => d.Date).getTime()
  let max_ts = d3.max(Res_Elv, d => d.Date).getTime()

  // Selection state lives here and survives bounds updates
  // Set starting slider time to rough start of drought
  const default_start = Date.UTC(2000, 0, 1)   // 2000-01-01 in UTC
  let start_ts = Math.min(Math.max(default_start, min_ts), max_ts - 86400000)
  // Set slider end to max date
  let end_ts   = max_ts

  const fmt = ts => new Date(ts).toLocaleDateString("en-US", {
    month: "short", day: "numeric", year: "numeric", timeZone: "UTC"
  })

  // Clamp to 0-100 so a handle never leaves the track if bounds shrink past it
  const to_pct = ts => {
    if (max_ts === min_ts) return 0
    return Math.max(0, Math.min(100, ((ts - min_ts) / (max_ts - min_ts)) * 100))
  }

  const container = html`
    <div style="
      font-family: sans-serif;
      font-size: 12px;
      padding: 8px 0 4px 0;
      width: 100%;
      user-select: none;
    ">
      <div style="display: flex; justify-content: space-between; margin-bottom: 6px; color: #6c757d;">
        <span id="start-label">${fmt(start_ts)}</span>
        <span id="end-label">${fmt(end_ts)}</span>
      </div>

      <div id="track" style="
        position: relative;
        height: 6px;
        background: #dee2e6;
        border-radius: 3px;
        margin: 10px 20px 10px 40px;
        cursor: pointer;
      ">
        <div id="range-fill" style="
          position: absolute;
          height: 100%;
          background: steelblue;
          border-radius: 3px;
          left: 0%;
          right: 0%;
        "></div>

        <div id="handle-start" style="
          position: absolute;
          width: 16px;
          height: 16px;
          background: white;
          border: 2px solid steelblue;
          border-radius: 50%;
          top: 50%;
          transform: translate(-50%, -50%);
          cursor: ew-resize;
          left: 0%;
          box-shadow: 0 1px 4px rgba(0,0,0,0.2);
          z-index: 2;
        "></div>

        <div id="handle-end" style="
          position: absolute;
          width: 16px;
          height: 16px;
          background: white;
          border: 2px solid steelblue;
          border-radius: 50%;
          top: 50%;
          transform: translate(-50%, -50%);
          cursor: ew-resize;
          left: 100%;
          box-shadow: 0 1px 4px rgba(0,0,0,0.2);
          z-index: 2;
        "></div>
      </div>

      <div style="display: flex; justify-content: space-between; color: #aaaaaa; font-size: 11px;">
        <span id="min-label">${fmt(min_ts)}</span>
        <span id="max-label">${fmt(max_ts)}</span>
      </div>
    </div>
  `

  const track        = container.querySelector("#track")
  const handle_start = container.querySelector("#handle-start")
  const handle_end   = container.querySelector("#handle-end")
  const range_fill   = container.querySelector("#range-fill")
  const start_label  = container.querySelector("#start-label")
  const end_label    = container.querySelector("#end-label")
  const min_label    = container.querySelector("#min-label")
  const max_label    = container.querySelector("#max-label")

  const update_visuals = () => {
    const s_pct = to_pct(start_ts)
    const e_pct = to_pct(end_ts)
    handle_start.style.left = s_pct + "%"
    handle_end.style.left   = e_pct + "%"
    range_fill.style.left   = s_pct + "%"
    range_fill.style.right  = (100 - e_pct) + "%"
    start_label.textContent = fmt(start_ts)
    end_label.textContent   = fmt(end_ts)
    min_label.textContent   = fmt(min_ts)
    max_label.textContent   = fmt(max_ts)
  }

  const mouse_to_ts = (e) => {
    const rect = track.getBoundingClientRect()
    const pct  = Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width))
    return min_ts + pct * (max_ts - min_ts)
  }

  const snap_to_day = ts => {
    const d = new Date(ts)
    d.setUTCHours(0, 0, 0, 0)
    return d.getTime()
  }

  const make_draggable = (handle, is_start) => {
    handle.addEventListener("mousedown", e => {
      e.preventDefault()

      const on_move = e => {
        const ts = snap_to_day(mouse_to_ts(e))
        if (is_start) {
          start_ts = Math.min(ts, end_ts - 86400000)
        } else {
          end_ts = Math.max(ts, start_ts + 86400000)
        }
        update_visuals()
        container.value = [new Date(start_ts), new Date(end_ts)]
        container.dispatchEvent(new Event("input"))
      }

      const on_up = () => {
        window.removeEventListener("mousemove", on_move)
        window.removeEventListener("mouseup", on_up)
      }

      window.addEventListener("mousemove", on_move)
      window.addEventListener("mouseup", on_up)
    })
  }

  make_draggable(handle_start, true)
  make_draggable(handle_end,   false)

  // NEW: update the track bounds only; selection is left alone
  // and no "input" event is fired because the value hasn't changed
  container.setBounds = (new_min, new_max) => {
    min_ts = new_min.getTime()
    max_ts = new_max.getTime()
    update_visuals()
  }

  update_visuals()
  container.value = [new Date(start_ts), new Date(end_ts)]

  return container
}
