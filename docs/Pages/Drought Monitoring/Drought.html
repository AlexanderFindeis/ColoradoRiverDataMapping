{
  // Define custom color mapping
  const drought_color_map = {
    "0": "#FFFF00",
    "1": "#FCD37F", 
    "2": "#FFAA00",
    "3": "#E60000",
    "4": "#730000"
  }

  // Legend labeling
  const drought_label_map = {
    "0": "D0 - Abnormally Dry",
    "1": "D1 - Moderate Drought",
    "2": "D2 - Severe Drought",
    "3": "D3 - Extreme Drought",
    "4": "D4 - Exceptional Drought"
  }

  // Map container
  let svg = geoviz.create({ 
    domain:   Basin_States, 
    width:    width, 
    margin: [25, 25, 60, 25],
    zoomable: [0.5, 5],
    background: "#ffffff00",
    outline:  false
  })

  svg.path({
    strokeWidth: 0.2,
    data:        drought_layer,
    fill:        d => drought_color_map[String(d.properties.DM)] ?? "#cccccc",
    tip:         d => drought_label_map[String(d.properties.DM)] ?? "No Data"
  })

  // Legend
  svg.legend.typo_vertical({
    pos:         [20, 500],
    title:       "Drought Level",
    alphabetical: false,
    colors:      Object.values(drought_color_map),
    types:       Object.values(drought_label_map),
    missing:     false
  })

  svg.path({
    id: "basin-states",
    datum:       Basin_States,
    fill:        "#00000000",
    stroke:      "#000000",
    strokeWidth: 2,
    tip: false,
    styles:      "pointer-events: none;"
  })

  svg.path({
    id:          "upper-basin",
    datum:       Upper_Basin,
    fill:        "#1a64960a",
    stroke:      "#1a6496",
    strokeWidth: 2,
    tip: false,
    styles:      "pointer-events: none;"
  })

  svg.path({
    id:          "lower-basin",
    datum:       Lower_Basin,
    fill:        "#a944420a",
    stroke:      "#a94442",
    strokeWidth: 2,
    tip: false,
    styles:      "pointer-events: none;"
  })

  return svg.render()
}
