Res_Elv = (await d3.csv("https://data.coloradoriveratlas.org/ReservoirData/Reservoir_Elevation.csv")).map(d => ({
  ...d,
  Date: new Date(d.Date + "T00:00:00")
})).filter(d => d.Type === "Recorded")
render_date = d3.max(Res_Elv, d => d.Date).toLocaleDateString("en-US", {
  month: "long",
  day: "numeric",
  year: "numeric"
})

<div style="
  background: #5cabfa2f;

  color: #6c757d;
  margin-bottom: 16px;
  text-align: center;
">
  Data current as of <b>${render_date}</b>

`
